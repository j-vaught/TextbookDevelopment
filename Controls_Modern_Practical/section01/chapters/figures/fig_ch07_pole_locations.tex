% Figure: Pole Locations in S-Plane for Different Damping
% Chapter 7: Time-Domain Analysis
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=0.95]
    % Axes
    \draw[thick, UofSCBlack, ->] (-6, 0) -- (1.5, 0) node[right] {$\sigma$ (Real)};
    \draw[thick, UofSCBlack, ->] (0, -4.5) -- (0, 4.5) node[above] {$j\omega$ (Imaginary)};

    % Grid
    \draw[UofSC50Black!40, very thin] (-6, -4) grid[step=1] (1, 4);

    % Natural frequency circles (omega_n = 2)
    \draw[UofSC70Black, dashed, thin] (0, 0) circle (2);
    \node[UofSC70Black, font=\tiny] at (1.5, 1.5) {$\omega_n$};

    % Constant damping ratio lines
    % zeta = 0.3 line (angle = acos(0.3) = 72.5 deg)
    \draw[UofSCRose!50, thin, dashed] (0, 0) -- (-5, 4.77);
    \draw[UofSCRose!50, thin, dashed] (0, 0) -- (-5, -4.77);

    % zeta = 0.7 line (angle = acos(0.7) = 45.6 deg)
    \draw[UofSCAtlantic!50, thin, dashed] (0, 0) -- (-5, 5.1);
    \draw[UofSCAtlantic!50, thin, dashed] (0, 0) -- (-5, -5.1);

    % ===== UNDERDAMPED (zeta = 0.3) =====
    % sigma = -zeta*omega_n = -0.6, omega_d = omega_n*sqrt(1-zeta^2) = 1.91
    \fill[UofSCRose] (-0.6, 1.91) circle (0.15);
    \fill[UofSCRose] (-0.6, -1.91) circle (0.15);
    \draw[UofSCRose, thick] (-0.6, 1.91) -- (-0.6, -1.91);
    \node[UofSCRose, font=\small, right] at (-0.4, 2.2) {$\zeta = 0.3$};

    % ===== UNDERDAMPED (zeta = 0.7) =====
    % sigma = -1.4, omega_d = 1.43
    \fill[UofSCAtlantic] (-1.4, 1.43) circle (0.15);
    \fill[UofSCAtlantic] (-1.4, -1.43) circle (0.15);
    \draw[UofSCAtlantic, thick] (-1.4, 1.43) -- (-1.4, -1.43);
    \node[UofSCAtlantic, font=\small, above left] at (-1.5, 1.6) {$\zeta = 0.7$};

    % ===== CRITICALLY DAMPED (zeta = 1.0) =====
    % Repeated pole at sigma = -omega_n = -2
    \fill[UofSCHorseshoe] (-2, 0) circle (0.18);
    \node[UofSCHorseshoe, font=\footnotesize] at (-2, 0.15) {$\times 2$};
    \node[UofSCHorseshoe, font=\small, below] at (-2, -0.4) {$\zeta = 1.0$};

    % ===== OVERDAMPED (zeta = 1.5) =====
    % Two real poles: s = -omega_n*(zeta +/- sqrt(zeta^2-1))
    % s1 = -2*(1.5 + 1.118) = -5.24, s2 = -2*(1.5 - 1.118) = -0.76
    \fill[UofSCCongaree] (-5.24, 0) circle (0.15);
    \fill[UofSCCongaree] (-0.76, 0) circle (0.15);
    \draw[UofSCCongaree, thick, <->] (-5.24, 0.3) -- (-0.76, 0.3);
    \node[UofSCCongaree, font=\small, above] at (-3, 0.5) {$\zeta = 1.5$};

    % Stability boundary annotation
    \fill[UofSCGarnet!10] (0, -4) rectangle (1.5, 4);
    \node[UofSCGarnet, font=\footnotesize, rotate=90] at (0.8, 0) {Unstable Region};

    % Annotations with arrows
    \draw[->, UofSCRose, thick] (-0.6, 3) -- (-0.6, 2.1);
    \node[UofSCRose, font=\footnotesize, align=center, above] at (-0.6, 3) {Oscillatory\\response};

    \draw[->, UofSCHorseshoe, thick] (-3.5, -1.5) -- (-2.15, -0.15);
    \node[UofSCHorseshoe, font=\footnotesize, align=center, below] at (-3.5, -1.5) {Fastest non-\\oscillatory};

    \draw[->, UofSCCongaree, thick] (-5.24, -1.2) -- (-5.24, -0.2);
    \node[UofSCCongaree, font=\footnotesize, align=center, below] at (-4.5, -1.5) {Dominant pole\\(slow mode)};

    % Dimension annotations
    \draw[<->, UofSC70Black, thin] (-0.6, -2.5) -- (0, -2.5);
    \node[UofSC70Black, font=\tiny, below] at (-0.3, -2.5) {$\sigma = \zeta\omega_n$};

    \draw[<->, UofSC70Black, thin] (0.3, 0) -- (0.3, 1.91);
    \node[UofSC70Black, font=\tiny, right] at (0.35, 1) {$\omega_d$};

\end{tikzpicture}
\caption{Pole locations in the $s$-plane for a second-order system with $\omega_n = 2$ rad/s and various damping ratios. Underdamped systems have complex conjugate poles, critically damped has a repeated real pole, and overdamped has two distinct real poles.}
\label{fig:pole_locations}
\end{figure}
