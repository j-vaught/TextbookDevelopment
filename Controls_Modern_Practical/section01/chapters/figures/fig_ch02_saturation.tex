% Figure: Saturation Nonlinearity
% Chapter 2: Control Systems Framework
\begin{figure}[htbp]
\centering
\begin{tikzpicture}
    \begin{axis}[
        width=10cm,
        height=7cm,
        xlabel={Input $u$},
        ylabel={Output $y = \mathrm{sat}(u)$},
        xmin=-3, xmax=3,
        ymin=-2, ymax=2,
        xtick={-2, -1, 0, 1, 2},
        ytick={-1.5, -1, -0.5, 0, 0.5, 1, 1.5},
        xticklabels={$-u_{\max}$, $-1$, $0$, $1$, $u_{\max}$},
        yticklabels={$$, $-y_{\max}$, $$, $0$, $$, $y_{\max}$, $$},
        axis lines=middle,
        grid=major,
        grid style={UofSC50Black!30},
        every axis x label/.style={at={(ticklabel* cs:1.02)}, anchor=west},
        every axis y label/.style={at={(ticklabel* cs:1.02)}, anchor=south},
    ]
        % Linear region (unsaturated)
        \addplot[UofSCAtlantic, very thick, domain=-2:2] {x/2};

        % Saturation regions
        \addplot[UofSCGarnet, very thick, domain=-3:-2] {-1};
        \addplot[UofSCGarnet, very thick, domain=2:3] {1};

        % Ideal linear response (dashed)
        \addplot[UofSC70Black, dashed, thin, domain=-3:3] {x/2};

        % Annotation arrows and labels
        \node[UofSCAtlantic, font=\small, align=center] at (axis cs:0.8, -0.8) {Linear\\region};
        \node[UofSCGarnet, font=\small, align=center] at (axis cs:2.5, 1.4) {Positive\\saturation};
        \node[UofSCGarnet, font=\small, align=center] at (axis cs:-2.5, -1.4) {Negative\\saturation};

        % Saturation limits with horizontal dashed lines
        \addplot[UofSCRose, dashed, thin] coordinates {(-3, 1) (3, 1)};
        \addplot[UofSCRose, dashed, thin] coordinates {(-3, -1) (3, -1)};

        % Corner markers
        \fill[UofSCHorseshoe] (axis cs:-2, -1) circle (3pt);
        \fill[UofSCHorseshoe] (axis cs:2, 1) circle (3pt);

    \end{axis}

    % Mathematical definition box
    \node[draw=UofSC70Black, fill=UofSCSandstorm, align=left, font=\small,
          anchor=north west, sharp corners] at (8.5, 5.5) {
        $\mathrm{sat}(u) = \begin{cases}
            y_{\max} & u > u_{\max} \\
            Ku & |u| \leq u_{\max} \\
            -y_{\max} & u < -u_{\max}
        \end{cases}$
    };

\end{tikzpicture}
\caption{Saturation nonlinearity showing linear operating region and saturation limits. The slope $K$ in the linear region represents the actuator gain. Beyond the saturation limits $\pm u_{\max}$, the output remains constant at $\pm y_{\max}$.}
\label{fig:saturation}
\end{figure}
