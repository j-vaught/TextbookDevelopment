    % Axes
    \draw[thick, UofSCBlack] (-5,0) -- (1,0) node[right] {$\sigma$ (Real)};
    \draw[thick, UofSCBlack] (0,-4) -- (0,4) node[above] {$j\omega$ (Imaginary)};

    % Grid
    \draw[UofSC50Black, very thin] (-5,-4) grid (1,4);

    % Natural frequency reference circles (faint)
    \draw[UofSC70Black, dashed, thin] (0,0) circle (1);
    \draw[UofSC70Black, dashed, thin] (0,0) circle (2);
    \draw[UofSC70Black, dashed, thin] (0,0) circle (3);

    % Underdamped case (zeta < 1): poles off real axis
    % For omega_n = 3, zeta = 0.3
    % sigma = -0.9, omega_d = 2.86
    \fill[UofSCRose] (-0.9, 2.86) circle (0.12);
    \fill[UofSCRose] (-0.9, -2.86) circle (0.12);
    \node[UofSCRose, font=\small, above] at (-0.9, 3.1) {$\zeta=0.3$};

    % Underdamped case: zeta = 0.6, omega_n = 3
    % sigma = -1.8, omega_d = 2.4
    \fill[UofSCAtlantic] (-1.8, 2.4) circle (0.12);
    \fill[UofSCAtlantic] (-1.8, -2.4) circle (0.12);
    \node[UofSCAtlantic, font=\small, above] at (-1.8, 2.65) {$\zeta=0.6$};

    % Critically damped case (zeta = 1): repeated pole on real axis
    % For omega_n = 3, zeta = 1.0
    % sigma = -3, omega_d = 0
    \fill[UofSCHorseshoe] (-3, 0) circle (0.14);
    \node[UofSCHorseshoe, font=\small, below] at (-3, -0.35) {$\zeta=1.0$};

    % Overdamped case (zeta > 1): two real poles
    % For omega_n = 3, zeta = 1.5
    % poles at -4.5 and -1.5
    \fill[UofSCCongaree] (-4.5, 0) circle (0.12);
    \fill[UofSCCongaree] (-1.5, 0) circle (0.12);
    \node[UofSCCongaree, font=\small, below] at (-3, -0.6) {$\zeta=1.5$};

    % Stability boundary (imaginary axis)
    \draw[UofSCGarnet, line width=1pt] (0,-4) -- (0,4);

    % Response region labels
    \node[UofSCRose, font=\footnotesize, align=center] at (-2.5, 3.5) {Underdamped\\oscillatory};
    \node[UofSCHorseshoe, font=\footnotesize, align=center] at (-3.5, -1.2) {Critically\\damped};
    \node[UofSCCongaree, font=\footnotesize, align=center] at (-4.5, 1.5) {Overdamped};

\end{tikzpicture}
\caption{Pole-zero map showing pole locations for different damping ratios ($\omega_n = 3$ rad/s fixed). As $\zeta$ increases from underdamped to overdamped, poles move from the complex plane onto the real axis.}
