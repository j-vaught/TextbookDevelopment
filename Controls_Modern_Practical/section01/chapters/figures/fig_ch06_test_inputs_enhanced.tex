\begin{tikzpicture}
    % ========== IMPULSE (Top Left) ==========
    \begin{scope}[shift={(0, 5)}]
        \node[font=\bfseries\small, UofSCGarnet] at (1.5, 2.3) {Impulse};

        \begin{axis}[
            at={(0,0)},
            width=5cm, height=3.5cm,
            xlabel={$t$}, ylabel={$\delta(t)$},
            xlabel style={font=\footnotesize},
            ylabel style={font=\footnotesize},
            xmin=-0.5, xmax=3,
            ymin=-0.2, ymax=1.5,
            xtick={0, 1, 2},
            ytick={0, 1},
            axis lines=middle,
            grid=none,
            every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west, font=\footnotesize},
            every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south, font=\footnotesize},
        ]
            % Impulse arrow
            \draw[->, UofSCGarnet, line width=2pt] (axis cs:0, 0) -- (axis cs:0, 1.2);
            \node[UofSCGarnet, font=\footnotesize] at (axis cs:0.3, 1.3) {$1$};

            % Zero elsewhere
            \addplot[UofSCAtlantic, thick, domain=-0.5:0] {0};
            \addplot[UofSCAtlantic, thick, domain=0.01:3] {0};
        \end{axis}

        % Laplace transform
        \node[draw=UofSC70Black, fill=UofSCSandstorm, font=\footnotesize,
              sharp corners, inner sep=3pt] at (3.8, 0.8) {$\mathcal{L}\{\delta(t)\} = 1$};
    \end{scope}

    % ========== STEP (Top Right) ==========
    \begin{scope}[shift={(6.5, 5)}]
        \node[font=\bfseries\small, UofSCGarnet] at (1.5, 2.3) {Unit Step};

        \begin{axis}[
            at={(0,0)},
            width=5cm, height=3.5cm,
            xlabel={$t$}, ylabel={$u(t)$},
            xlabel style={font=\footnotesize},
            ylabel style={font=\footnotesize},
            xmin=-0.5, xmax=3,
            ymin=-0.2, ymax=1.5,
            xtick={0, 1, 2},
            ytick={0, 1},
            axis lines=middle,
            grid=none,
            every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west, font=\footnotesize},
            every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south, font=\footnotesize},
        ]
            % Step function
            \addplot[UofSCAtlantic, thick, domain=-0.5:0] {0};
            \addplot[UofSCHorseshoe, very thick, domain=0:3] {1};

            % Discontinuity marker
            \fill[UofSCHorseshoe] (axis cs:0, 1) circle (2pt);
            \draw[UofSCAtlantic, thick] (axis cs:0, 0) circle (2pt);
        \end{axis}

        % Laplace transform
        \node[draw=UofSC70Black, fill=UofSCSandstorm, font=\footnotesize,
              sharp corners, inner sep=3pt] at (3.8, 0.8) {$\mathcal{L}\{u(t)\} = \dfrac{1}{s}$};
    \end{scope}

    % ========== RAMP (Bottom Left) ==========
    \begin{scope}[shift={(0, 0)}]
        \node[font=\bfseries\small, UofSCGarnet] at (1.5, 2.3) {Ramp};

        \begin{axis}[
            at={(0,0)},
            width=5cm, height=3.5cm,
            xlabel={$t$}, ylabel={$r(t) = tu(t)$},
            xlabel style={font=\footnotesize},
            ylabel style={font=\footnotesize},
            xmin=-0.5, xmax=3,
            ymin=-0.5, ymax=3,
            xtick={0, 1, 2},
            ytick={0, 1, 2},
            axis lines=middle,
            grid=none,
            every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west, font=\footnotesize},
            every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south, font=\footnotesize},
        ]
            % Ramp function
            \addplot[UofSCAtlantic, thick, domain=-0.5:0] {0};
            \addplot[UofSCCongaree, very thick, domain=0:2.8] {x};

            % Origin marker
            \fill[UofSCCongaree] (axis cs:0, 0) circle (2pt);
        \end{axis}

        % Laplace transform
        \node[draw=UofSC70Black, fill=UofSCSandstorm, font=\footnotesize,
              sharp corners, inner sep=3pt] at (3.8, 0.8) {$\mathcal{L}\{tu(t)\} = \dfrac{1}{s^2}$};
    \end{scope}

    % ========== SINUSOID (Bottom Right) ==========
    \begin{scope}[shift={(6.5, 0)}]
        \node[font=\bfseries\small, UofSCGarnet] at (1.5, 2.3) {Sinusoid};

        \begin{axis}[
            at={(0,0)},
            width=5cm, height=3.5cm,
            xlabel={$t$}, ylabel={$\sin(\omega t)$},
            xlabel style={font=\footnotesize},
            ylabel style={font=\footnotesize},
            xmin=-0.5, xmax=3,
            ymin=-1.5, ymax=1.5,
            xtick={0, 1.57, 3.14},
            xticklabels={$0$, $\frac{\pi}{\omega}$, $\frac{2\pi}{\omega}$},
            ytick={-1, 0, 1},
            axis lines=middle,
            grid=none,
            every axis x label/.style={at={(ticklabel* cs:1)}, anchor=west, font=\footnotesize},
            every axis y label/.style={at={(ticklabel* cs:1)}, anchor=south, font=\footnotesize},
        ]
            % Sinusoid
            \addplot[UofSCRose, very thick, domain=0:3, samples=100] {sin(deg(2*x))};
            \addplot[UofSCAtlantic, thick, domain=-0.5:0] {0};

            % Origin marker
            \fill[UofSCRose] (axis cs:0, 0) circle (2pt);
        \end{axis}

        % Laplace transform
        \node[draw=UofSC70Black, fill=UofSCSandstorm, font=\footnotesize,
              sharp corners, inner sep=3pt] at (4.0, 0.8) {$\mathcal{L}\{\sin\omega t\} = \dfrac{\omega}{s^2+\omega^2}$};
    \end{scope}

\end{tikzpicture}
\caption{The four standard test inputs used in control systems analysis: impulse $\delta(t)$, unit step $u(t)$, ramp $r(t) = tu(t)$, and sinusoid $\sin(\omega t)$. Each signal is shown with its corresponding Laplace transform for frequency-domain analysis.}
\label{fig:test_inputs_enhanced}
\end{figure}
