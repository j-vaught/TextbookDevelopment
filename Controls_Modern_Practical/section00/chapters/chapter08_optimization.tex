\chapter{Optimization}
\label{ch:optimization}
% ============================================================================

Optimization is central to optimal control, parameter estimation, and machine learning.

\section{Unconstrained Optimization}

\subsubsection{Necessary Condition}
At a local minimum $\vecx^*$:
\begin{equation}
\nabla f(\vecx^*) = \mathbf{0}
\end{equation}

\subsubsection{Sufficient Condition}
Also $\mH(\vecx^*) \succ 0$ (positive definite Hessian).

\subsubsection{Gradient Descent}
\begin{equation}
\vecx_{k+1} = \vecx_k - \alpha \nabla f(\vecx_k)
\end{equation}
where $\alpha > 0$ is the step size (learning rate).

\subsubsection{Newton's Method}
\begin{equation}
\vecx_{k+1} = \vecx_k - [\mH(\vecx_k)]^{-1} \nabla f(\vecx_k)
\end{equation}

\section{Constrained Optimization}

\subsubsection{Equality Constraints}
Minimize $f(\vecx)$ subject to $\vecg(\vecx) = \mathbf{0}$

The \textbf{Lagrangian}:
\begin{equation}
L(\vecx, \boldsymbol{\lambda}) = f(\vecx) + \boldsymbol{\lambda}^\top \vecg(\vecx)
\end{equation}

\subsubsection{Necessary Conditions (KKT)}
\begin{align}
\nabla_\vecx L &= \mathbf{0} \\
\nabla_{\boldsymbol{\lambda}} L &= \vecg(\vecx) = \mathbf{0}
\end{align}

\subsubsection{Inequality Constraints}
Minimize $f(\vecx)$ subject to $\vech(\vecx) \leq \mathbf{0}$

The \textbf{Karush-Kuhn-Tucker (KKT) conditions}:
\begin{align}
\nabla f(\vecx^*) + \sum_i \mu_i \nabla h_i(\vecx^*) &= \mathbf{0} \\
h_i(\vecx^*) &\leq 0 \\
\mu_i &\geq 0 \\
\mu_i h_i(\vecx^*) &= 0 \quad \text{(complementary slackness)}
\end{align}

\section{Convex Optimization}

\begin{definitionbox}[title=Definition 0.7: Convex Function]
A function $f: \mathbb{R}^n \to \mathbb{R}$ is \textbf{convex} if for all $\vecx, \vecy$ and $\theta \in [0,1]$:
\begin{equation}
f(\theta\vecx + (1-\theta)\vecy) \leq \theta f(\vecx) + (1-\theta)f(\vecy)
\end{equation}
\end{definitionbox}

Key property: Any local minimum of a convex function is a global minimum.

\subsubsection{Examples of Convex Problems}
\begin{itemize}
    \item Linear programming: $\min \vecc^\top\vecx$ s.t. $\mA\vecx \leq \vecb$
    \item Quadratic programming: $\min \frac{1}{2}\vecx^\top\mQ\vecx + \vecc^\top\vecx$ s.t. $\mA\vecx \leq \vecb$ (if $\mQ \succeq 0$)
    \item Least squares: $\min \|\mA\vecx - \vecb\|^2$
\end{itemize}

\section{Dynamic Programming}

For sequential decision problems, \textbf{Bellman's Principle of Optimality}:

\begin{quote}
An optimal policy has the property that whatever the initial state and initial decision are, the remaining decisions must constitute an optimal policy with regard to the state resulting from the first decision.
\end{quote}

The \textbf{Bellman equation} for discrete-time optimal control:
\begin{equation}
V^*(x_k) = \min_{u_k} \left[ L(x_k, u_k) + V^*(f(x_k, u_k)) \right]
\end{equation}

where $V^*$ is the optimal value function (cost-to-go).
