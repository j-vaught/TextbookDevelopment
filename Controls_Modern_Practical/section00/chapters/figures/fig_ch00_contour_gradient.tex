\centering
\begin{tikzpicture}
    \begin{axis}[
        width=10cm,
        height=8cm,
        xlabel={$x_1$},
        ylabel={$x_2$},
        xmin=-3, xmax=3,
        ymin=-3, ymax=3,
        view={0}{90},
        colormap={uofsc}{color(0cm)=(UofSCAtlantic!20); color(1cm)=(UofSCAtlantic!40); color(2cm)=(UofSCAtlantic!60); color(3cm)=(UofSCAtlantic!80); color(4cm)=(UofSCGarnet)},
        axis equal image,
    ]
        % Contour plot of quadratic function f(x,y) = x^2 + 4y^2
        \addplot[
            contour lua={
                number=8,
                labels=false,
            },
            thick,
            UofSCAtlantic,
            domain=-3:3,
            y domain=-3:3,
            samples=40,
        ] {x^2 + 4*y^2};

        % Minimum point
        \addplot[only marks, mark=*, mark size=4pt, UofSCGarnet] coordinates {(0,0)};
        \node[above right, UofSCGarnet, font=\small\bfseries] at (axis cs:0.15,0.15) {$\vecx^*$};

        % Gradient vectors at several points
        % At (2, 1): gradient = (4, 8), normalized and scaled
        \draw[->,UofSCGarnet,very thick] (axis cs:2,1) -- (axis cs:2-0.8,1-1.6);
        \node[above right, UofSCGarnet, font=\scriptsize] at (axis cs:2,1) {$-\nabla f$};

        % At (-1.5, 0.8): gradient = (-3, 6.4), normalized and scaled
        \draw[->,UofSCGarnet,very thick] (axis cs:-1.5,0.8) -- (axis cs:-1.5+0.6,0.8-1.3);

        % At (0.5, -1.2): gradient = (1, -9.6), normalized and scaled
        \draw[->,UofSCGarnet,very thick] (axis cs:0.5,-1.2) -- (axis cs:0.5-0.2,-1.2+1.9);

        % At (-2, -0.5): gradient = (-4, -4), normalized and scaled
        \draw[->,UofSCGarnet,very thick] (axis cs:-2,-0.5) -- (axis cs:-2+0.7,-0.5+0.7);

        % Level set labels
        \node[UofSCAtlantic, font=\scriptsize] at (axis cs:1.8,-0.3) {$f = 4$};
        \node[UofSCAtlantic, font=\scriptsize] at (axis cs:2.5,-0.5) {$f = 8$};

    \end{axis}
\end{tikzpicture}
\caption{Contour plot of $f(x_1, x_2) = x_1^2 + 4x_2^2$ showing level curves (lines of constant function value). The negative gradient vectors (red arrows) point toward the minimum $\vecx^* = (0, 0)$ and are perpendicular to the level curves. Elliptical contours indicate different scaling along each axis (condition number $\kappa = 4$), which causes gradient descent to take an inefficient zigzag path.}
\label{fig:contour_gradient}
\end{figure}
