    \addplot[UofSCGarnet, very thick, domain=0:5, samples=100] {2*exp(-x)};
    \addlegendentry{$x_1(t) = x_1(0)\ee^{-t}$, $\lambda_1 = -1$}

    \addplot[UofSCAtlantic, very thick, domain=0:5, samples=100] {1.5*exp(-2*x)};

    \addplot[UofSC50Black, dashed, thick] coordinates {(0,0) (5,0)};

    \draw[UofSCGarnet, thick, dashed] (axis cs:1,0) -- (axis cs:1,{2*exp(-1)});

    \draw[UofSCAtlantic, thick, dashed] (axis cs:0.5,0) -- (axis cs:0.5,{1.5*exp(-1)});

        $\vecx(t) = \ee^{\mA t}\vecx(0)$\\[4pt]
        Time constant: $\tau_i = -1/\Real(\lambda_i)$\\[2pt]
        Stable if $\Real(\lambda_i) < 0$
    };
\end{tikzpicture}
\caption{Time response of a second-order diagonal system governed by $\ee^{\mA t}$. Each state $x_i(t)$ decays exponentially with rate determined by its eigenvalue $\lambda_i$. The time constant $\tau_i = -1/\lambda_i$ indicates when the response reaches $\ee^{-1} \approx 37\%$ of its initial value. Both eigenvalues are negative (stable), with $\lambda_2 = -2$ causing faster decay than $\lambda_1 = -1$.}
\label{fig:matrix_exponential}
