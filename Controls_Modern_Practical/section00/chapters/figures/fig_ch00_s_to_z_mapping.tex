% ========== Mapping rules summary ==========
\begin{tikzpicture}
\node[draw=UofSCCongaree, thick, fill=UofSCCongaree!8, minimum width=14cm, minimum height=3.5cm, sharp corners, inner sep=10pt] {
    \begin{minipage}{13.5cm}
    \textbf{Key Mapping Properties} ($z = \ee^{sT}$ where $T$ is the sampling period, $\omega_s = 2\pi/T$):

    \vspace{0.3cm}
    \begin{tabular}{@{}lll@{}}
    \textbf{$s$-Plane Feature} & \textbf{$z$-Plane Feature} & \textbf{Implication} \\[3pt]
    Imaginary axis ($\sigma = 0$) & Unit circle ($|z| = 1$) & Stability boundary preserved \\[3pt]
    Left half-plane ($\sigma < 0$) & Inside unit circle ($|z| < 1$) & Stable poles map to stable poles \\[3pt]
    Right half-plane ($\sigma > 0$) & Outside unit circle ($|z| > 1$) & Unstable poles map to unstable poles \\[3pt]
    Vertical line ($\sigma = $ const) & Circle of radius $\ee^{\sigma T}$ & Decay rate $\to$ pole magnitude \\[3pt]
    Horizontal line ($\omega = $ const) & Ray at angle $\omega T$ & Frequency $\to$ angle on unit circle \\[3pt]
    $s = 0$ (DC) & $z = 1$ & Zero frequency \\[3pt]
    $s = \pm\jj\omega_s/2$ (Nyquist) & $z = -1$ & Highest representable frequency \\[3pt]
    Strip $-\omega_s/2 < \omega < \omega_s/2$ & Entire $z$-plane & \textbf{Frequency warping:} entire strip maps once \\
    \end{tabular}
    \end{minipage}
};
\end{tikzpicture}

\caption{Mapping from the continuous-time $s$-plane to the discrete-time $z$-plane via $z = \ee^{sT}$. \textbf{Left:} The $s$-plane with stable (Horseshoe) left half-plane and unstable (Rose) right half-plane. Vertical lines of constant $\sigma$ (Atlantic dashed) and horizontal lines of constant $\omega$ (Congaree dotted) are shown. \textbf{Right:} The $z$-plane where the imaginary axis maps to the unit circle (Garnet), vertical lines map to circles of constant magnitude (Atlantic dashed), and horizontal lines map to rays of constant angle (Congaree dotted). The primary strip $-\omega_s/2 < \omega < \omega_s/2$ in the $s$-plane maps to the entire $z$-plane, with higher frequency strips overlapping (aliasing effect). Sample poles (Atlantic dots) show how complex conjugate pairs transform.}
\label{fig:s_to_z_mapping}
\end{figure}
