\section{Unit Step Function}

\begin{equation}
u(t) = \begin{cases} 0 & t < 0 \\ 1 & t \geq 0 \end{cases}
\end{equation}

The discrete-time unit step:
\begin{equation}
u[n] = \begin{cases} 0 & n < 0 \\ 1 & n \geq 0 \end{cases}
\end{equation}

\section{Impulse (Dirac Delta) Function}

The \textbf{Dirac delta function} is defined by:
\begin{align}
\delta(t) &= 0 \quad \text{for } t \neq 0 \\
\int_{-\infty}^{\infty} \delta(t) \, \dd t &= 1
\end{align}

\subsubsection{Sifting property}
\begin{equation}
\int_{-\infty}^{\infty} f(t)\delta(t-t_0) \, \dd t = f(t_0)
\end{equation}

Relationship to step function:
\begin{equation}
\delta(t) = \frac{\dd u(t)}{\dd t}
\end{equation}

\section{Ramp Function}

\begin{equation}
r(t) = t \cdot u(t) = \begin{cases} 0 & t < 0 \\ t & t \geq 0 \end{cases}
\end{equation}

\section{Sinusoidal Functions}

General sinusoid:
\begin{equation}
x(t) = A\sin(\omega t + \phi) = A\cos(\omega t + \phi - \pi/2)
\end{equation}

where:
\begin{itemize}
    \item $A$ = amplitude
    \item $\omega = 2\pi f$ = angular frequency (rad/s)
    \item $f$ = frequency (Hz)
    \item $\phi$ = phase (rad)
    \item $T = 1/f = 2\pi/\omega$ = period (s)
\end{itemize}

\subsubsection{Complex exponential representation}
\begin{equation}
A\cos(\omega t + \phi) = \Real\{A\ee^{\jj(\omega t + \phi)}\}
\end{equation}

\section{Exponential Functions}

\subsubsection{Real exponential}
\begin{equation}
x(t) = A\ee^{\sigma t}
\end{equation}
\begin{itemize}
    \item $\sigma < 0$: decaying
    \item $\sigma > 0$: growing
    \item $\sigma = 0$: constant
\end{itemize}

\subsubsection{Complex exponential (damped sinusoid)}
\begin{equation}
x(t) = A\ee^{\sigma t}\cos(\omega t + \phi)
\end{equation}
