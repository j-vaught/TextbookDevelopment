\section{Numerical Integration of ODEs}

\subsubsection{Euler's Method (Forward)}
\begin{equation}
x_{n+1} = x_n + h f(t_n, x_n)
\end{equation}

\subsubsection{Runge-Kutta 4th Order (RK4)}
\begin{align}
k_1 &= f(t_n, x_n) \\
k_2 &= f(t_n + h/2, x_n + hk_1/2) \\
k_3 &= f(t_n + h/2, x_n + hk_2/2) \\
k_4 &= f(t_n + h, x_n + hk_3) \\
x_{n+1} &= x_n + \frac{h}{6}(k_1 + 2k_2 + 2k_3 + k_4)
\end{align}

\section{Root Finding}

\subsubsection{Newton-Raphson}
\begin{equation}
x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)}
\end{equation}

\subsubsection{Bisection Method}
For continuous $f$ with $f(a)f(b) < 0$, repeatedly halve the interval containing the root.

\section{Numerical Linear Algebra}

Key algorithms:
\begin{itemize}
    \item \textbf{Gaussian elimination}: $O(n^3)$ for solving $\mA\vecx = \vecb$
    \item \textbf{QR iteration}: Computing eigenvalues
    \item \textbf{SVD algorithms}: Golub-Kahan bidiagonalization
    \item \textbf{Iterative methods}: Conjugate gradient, GMRES for large sparse systems
\end{itemize}
