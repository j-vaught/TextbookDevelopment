\section{Differential Calculus}

\subsection{Derivatives and Rules}

The derivative of a function $f(t)$ represents its instantaneous rate of change:
\begin{equation}
\frac{\dd f}{\dd t} = f'(t) = \lim_{\Delta t \to 0} \frac{f(t + \Delta t) - f(t)}{\Delta t}
\end{equation}

\subsubsection{Key Differentiation Rules}

\begin{table}[htbp]
\centering
\begin{tabular}{ll}
\toprule
\textbf{Rule} & \textbf{Formula} \\
\midrule
Power Rule & $\frac{\dd}{\dd t}(t^n) = n t^{n-1}$ \\
Exponential & $\frac{\dd}{\dd t}(\ee^{at}) = a\ee^{at}$ \\
Logarithm & $\frac{\dd}{\dd t}(\ln t) = \frac{1}{t}$ \\
Sine & $\frac{\dd}{\dd t}(\sin\omega t) = \omega\cos\omega t$ \\
Cosine & $\frac{\dd}{\dd t}(\cos\omega t) = -\omega\sin\omega t$ \\
Product Rule & $\frac{\dd}{\dd t}(fg) = f'g + fg'$ \\
Quotient Rule & $\frac{\dd}{\dd t}\left(\frac{f}{g}\right) = \frac{f'g - fg'}{g^2}$ \\
Chain Rule & $\frac{\dd}{\dd t}f(g(t)) = f'(g(t)) \cdot g'(t)$ \\
\bottomrule
\end{tabular}
\caption{Essential differentiation rules for control systems.}
\label{tab:diff_rules}
\end{table}

\begin{examplebox}[title=Product Rule Application]
Differentiate $f(t) = t^2 \ee^{-at}$ where $a > 0$ is a constant.

\textbf{Solution:} Using the product rule with $u = t^2$ and $v = \ee^{-at}$:
\begin{align}
\frac{\dd f}{\dd t} &= \frac{\dd}{\dd t}(t^2) \cdot \ee^{-at} + t^2 \cdot \frac{\dd}{\dd t}(\ee^{-at}) \\
&= 2t \cdot \ee^{-at} + t^2 \cdot (-a\ee^{-at}) \\
&= \ee^{-at}(2t - at^2) \\
&= t\ee^{-at}(2 - at)
\end{align}
This type of function appears frequently in control systems as a damped response with a time-varying amplitude.
\end{examplebox}

\subsection{Partial Derivatives}

For functions of multiple variables $f(x_1, x_2, \ldots, x_n)$, partial derivatives hold all variables constant except one:
\begin{equation}
\frac{\partial f}{\partial x_i} = \lim_{\Delta x_i \to 0} \frac{f(x_1, \ldots, x_i + \Delta x_i, \ldots, x_n) - f(x_1, \ldots, x_n)}{\Delta x_i}
\end{equation}

The \textbf{gradient} is the vector of all partial derivatives:
\begin{equation}
\nabla f = \begin{bmatrix} \frac{\partial f}{\partial x_1} & \frac{\partial f}{\partial x_2} & \cdots & \frac{\partial f}{\partial x_n} \end{bmatrix}^\top
\end{equation}

The \textbf{Hessian} is the matrix of second partial derivatives:
\begin{equation}
\mH = \begin{bmatrix}
\frac{\partial^2 f}{\partial x_1^2} & \frac{\partial^2 f}{\partial x_1 \partial x_2} & \cdots \\
\frac{\partial^2 f}{\partial x_2 \partial x_1} & \frac{\partial^2 f}{\partial x_2^2} & \cdots \\
\vdots & \vdots & \ddots
\end{bmatrix}
\end{equation}

\subsection{Taylor Series Expansion}

Taylor series are essential for linearization of nonlinear systems:
\begin{equation}
f(x) = f(a) + f'(a)(x-a) + \frac{f''(a)}{2!}(x-a)^2 + \frac{f'''(a)}{3!}(x-a)^3 + \cdots
\end{equation}

For multivariate functions about point $\vecx_0$:
\begin{equation}
f(\vecx) \approx f(\vecx_0) + \nabla f(\vecx_0)^\top (\vecx - \vecx_0) + \frac{1}{2}(\vecx - \vecx_0)^\top \mH(\vecx_0) (\vecx - \vecx_0)
\end{equation}

\begin{keyconceptbox}[title=Linearization in Control]
For nonlinear systems $\dot{\vecx} = \vecf(\vecx, \vecu)$, we linearize about an equilibrium point $(\vecx_0, \vecu_0)$:
\begin{equation}
\Delta\dot{\vecx} \approx \mA \Delta\vecx + \mB \Delta\vecu
\end{equation}
where $\mA = \frac{\partial \vecf}{\partial \vecx}\big|_{(\vecx_0, \vecu_0)}$ and $\mB = \frac{\partial \vecf}{\partial \vecu}\big|_{(\vecx_0, \vecu_0)}$
\end{keyconceptbox}

\begin{examplebox}[title=Taylor Expansion of Sine Function]
Find the Taylor expansion of $\sin(x)$ about $x = 0$ (Maclaurin series) to fourth order.

\textbf{Solution:} We need the derivatives of $f(x) = \sin(x)$ evaluated at $x = 0$:
\begin{align}
f(x) &= \sin(x) & f(0) &= 0 \\
f'(x) &= \cos(x) & f'(0) &= 1 \\
f''(x) &= -\sin(x) & f''(0) &= 0 \\
f'''(x) &= -\cos(x) & f'''(0) &= -1 \\
f^{(4)}(x) &= \sin(x) & f^{(4)}(0) &= 0
\end{align}

Substituting into the Taylor series formula:
\begin{equation}
\sin(x) = 0 + 1 \cdot x + \frac{0}{2!}x^2 + \frac{-1}{3!}x^3 + \frac{0}{4!}x^4 + \cdots = x - \frac{x^3}{6} + \mathcal{O}(x^5)
\end{equation}

For small angles ($|x| \ll 1$), we often use the approximation $\sin(x) \approx x$, which is the first-order Taylor expansion. This is the ``small angle approximation'' commonly used in linearizing pendulum and rotational dynamics.
\end{examplebox}

\section{Integral Calculus}

\subsection{Definite and Indefinite Integrals}

The indefinite integral (antiderivative):
\begin{equation}
\int f(t) \, \dd t = F(t) + C \quad \text{where} \quad F'(t) = f(t)
\end{equation}

The definite integral:
\begin{equation}
\int_a^b f(t) \, \dd t = F(b) - F(a)
\end{equation}

\subsection{Integration by Parts}

\begin{equation}
\int u \, \dd v = uv - \int v \, \dd u
\end{equation}

This technique is essential for deriving Laplace transform properties.

\begin{examplebox}[title=Integration by Parts]
Evaluate the integral $\displaystyle\int t \ee^{-t} \, \dd t$.

\textbf{Solution:} Choose $u = t$ and $\dd v = \ee^{-t} \, \dd t$. Then:
\begin{align}
u &= t & \dd u &= \dd t \\
\dd v &= \ee^{-t} \, \dd t & v &= -\ee^{-t}
\end{align}

Applying integration by parts:
\begin{align}
\int t \ee^{-t} \, \dd t &= uv - \int v \, \dd u \\
&= t(-\ee^{-t}) - \int (-\ee^{-t}) \, \dd t \\
&= -t\ee^{-t} + \int \ee^{-t} \, \dd t \\
&= -t\ee^{-t} - \ee^{-t} + C \\
&= -\ee^{-t}(t + 1) + C
\end{align}

This integral appears when computing Laplace transforms of polynomial-exponential products.
\end{examplebox}

\subsection{Integration Techniques}

\subsubsection{Substitution (u-Substitution)}

When an integrand contains a composite function, substitution can simplify the integral:
\begin{equation}
\int f(g(t)) g'(t) \, \dd t = \int f(u) \, \dd u \quad \text{where } u = g(t)
\end{equation}

\begin{keyconceptbox}[title=Choosing a Substitution]
Look for an ``inner function'' $u = g(t)$ whose derivative $g'(t)$ also appears in the integrand. Common patterns include:
\begin{itemize}
    \item $\int \ee^{at} \cdot a \, \dd t$: let $u = at$
    \item $\int \frac{f'(t)}{f(t)} \, \dd t$: let $u = f(t)$, giving $\ln|f(t)| + C$
    \item $\int t^n f(t^{n+1}) \, \dd t$: let $u = t^{n+1}$
\end{itemize}
\end{keyconceptbox}

\subsubsection{Partial Fraction Decomposition}

For rational functions $\frac{P(s)}{Q(s)}$ where $\deg(P) < \deg(Q)$, partial fractions decompose the integrand into simpler terms. This is essential for inverse Laplace transforms.

\textbf{Case 1: Distinct real roots.} If $Q(s) = (s - p_1)(s - p_2)\cdots(s - p_n)$:
\begin{equation}
\frac{P(s)}{Q(s)} = \frac{A_1}{s - p_1} + \frac{A_2}{s - p_2} + \cdots + \frac{A_n}{s - p_n}
\end{equation}

\textbf{Case 2: Repeated real roots.} If $Q(s) = (s - p)^n$:
\begin{equation}
\frac{P(s)}{(s - p)^n} = \frac{A_1}{s - p} + \frac{A_2}{(s - p)^2} + \cdots + \frac{A_n}{(s - p)^n}
\end{equation}

\textbf{Case 3: Complex conjugate roots.} If $Q(s)$ contains $(s^2 + as + b)$ with $a^2 - 4b < 0$:
\begin{equation}
\frac{P(s)}{s^2 + as + b} = \frac{As + B}{s^2 + as + b}
\end{equation}

\begin{keyconceptbox}[title=Cover-Up Method for Residues]
For distinct real poles, the residue $A_k$ can be found by:
\begin{equation}
A_k = \lim_{s \to p_k} (s - p_k) \frac{P(s)}{Q(s)} = \frac{P(p_k)}{Q'(p_k)}
\end{equation}
This ``cover-up'' method is faster than solving systems of equations.
\end{keyconceptbox}

\subsection{Convolution Integral}

The convolution of two functions $f(t)$ and $g(t)$ is:
\begin{equation}
(f \conv g)(t) = \int_{-\infty}^{\infty} f(\tau) g(t - \tau) \, \dd\tau
\end{equation}

For causal systems (signals zero for $t < 0$):
\begin{equation}
(f \conv g)(t) = \int_0^t f(\tau) g(t - \tau) \, \dd\tau
\end{equation}

\begin{keyconceptbox}[title=Convolution in Control Systems]
The output $y(t)$ of a linear time-invariant (LTI) system with impulse response $h(t)$ to input $u(t)$ is:
\begin{equation}
y(t) = (h \conv u)(t) = \int_0^t h(\tau) u(t - \tau) \, \dd\tau
\end{equation}
\end{keyconceptbox}

\begin{examplebox}[title=Convolution of Two Exponentials]
Compute the convolution $(f \conv g)(t)$ for $t \geq 0$ where $f(t) = \ee^{-at}$ and $g(t) = \ee^{-bt}$ with $a \neq b$ and both signals are zero for $t < 0$.

\textbf{Solution:} Using the causal convolution formula:
\begin{align}
(f \conv g)(t) &= \int_0^t f(\tau) g(t - \tau) \, \dd\tau \\
&= \int_0^t \ee^{-a\tau} \ee^{-b(t-\tau)} \, \dd\tau \\
&= \ee^{-bt} \int_0^t \ee^{-a\tau} \ee^{b\tau} \, \dd\tau \\
&= \ee^{-bt} \int_0^t \ee^{(b-a)\tau} \, \dd\tau
\end{align}

Evaluating the integral:
\begin{align}
(f \conv g)(t) &= \ee^{-bt} \left[ \frac{\ee^{(b-a)\tau}}{b-a} \right]_0^t \\
&= \ee^{-bt} \cdot \frac{\ee^{(b-a)t} - 1}{b-a} \\
&= \frac{\ee^{-bt} \ee^{(b-a)t} - \ee^{-bt}}{b-a} \\
&= \frac{\ee^{-at} - \ee^{-bt}}{b-a}
\end{align}

This result is symmetric in the sense that swapping $a$ and $b$ gives the same answer (note the sign change in both numerator and denominator). This convolution appears when analyzing cascaded first-order systems.
\end{examplebox}
