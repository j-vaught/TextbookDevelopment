\chapter{Exercises}
\label{ch:math_exercises}
% ============================================================================

\begin{exercise}
Convert the following complex numbers to polar form:
\begin{enumerate}[label=(\alph*)]
    \item $z_1 = -3 + 4\jj$
    \item $z_2 = 2 - 2\jj$
    \item $z_3 = -5$
\end{enumerate}
\end{exercise}

\begin{exercise}
Find the Laplace transform of:
\begin{enumerate}[label=(\alph*)]
    \item $f(t) = 3\ee^{-2t}\sin(4t)$
    \item $f(t) = t^2\ee^{-t}$
    \item $f(t) = (1 - \ee^{-t})u(t)$
\end{enumerate}
\end{exercise}

\begin{exercise}
Find the inverse Laplace transform of:
\begin{enumerate}[label=(\alph*)]
    \item $F(s) = \frac{5}{s(s+3)}$
    \item $F(s) = \frac{s+1}{s^2+4s+13}$
    \item $F(s) = \frac{1}{(s+1)^2(s+2)}$
\end{enumerate}
\end{exercise}

\begin{exercise}
For the matrix $\mA = \begin{bmatrix} 0 & 1 \\ -2 & -3 \end{bmatrix}$:
\begin{enumerate}[label=(\alph*)]
    \item Find the eigenvalues
    \item Find the eigenvectors
    \item Compute $\ee^{\mA t}$
    \item Is the system $\dot{\vecx} = \mA\vecx$ stable?
\end{enumerate}
\end{exercise}

\begin{exercise}
Linearize the nonlinear system $\dot{x} = x^2 - u$ about the equilibrium point where $x_0 = 1$, $u_0 = 1$.
\end{exercise}

\begin{exercise}
Find the Z-transform and region of convergence for:
\begin{enumerate}[label=(\alph*)]
    \item $x[n] = (0.5)^n u[n]$
    \item $x[n] = n(0.8)^n u[n]$
    \item $x[n] = \cos(0.2\pi n)u[n]$
\end{enumerate}
\end{exercise}

\begin{exercise}
Using gradient descent, find the minimum of $f(x,y) = x^2 + 2y^2 - 2xy - 2x$ starting from $(0,0)$ with step size $\alpha = 0.1$. Perform 5 iterations.
\end{exercise}

\begin{exercise}
For a random variable $X \sim \mathcal{N}(5, 4)$:
\begin{enumerate}[label=(\alph*)]
    \item Find $P(X > 7)$
    \item Find $P(3 < X < 7)$
    \item Find the value $x$ such that $P(X < x) = 0.95$
\end{enumerate}
\end{exercise}
